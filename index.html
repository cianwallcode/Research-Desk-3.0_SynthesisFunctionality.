// 1. GLOBAL VARIABLES - Must be outside any functions let map; let selectedSite = null; let sites = []; // 2. Is database.json in the same folder?"); } } // 3. MAP LOGIC function initMap() { if (map) return; map = L.map('map', { zoomControl: false }).setView([53.3498, -6.2603], 7); L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png').addTo(map); sites.forEach(s => { if (s.coords && s.coords.length === 2) { let m = L.circleMarker(s.coords, { radius: 10, color: '#007aff', fillOpacity: 0.7 }).addTo(map); m.on('click', () => { selectedSite = s; updateSidebar(s); }); } }); } // 4. SIDEBAR UPDATE function updateSidebar(s) { const sideName = document.getElementById('side-name'); const sideInfo = document.getElementById('side-info'); if(sideName) sideName.innerText = s.name; if(sideInfo) { sideInfo.innerHTML = `

STYLE: ${s.style}

STATUS: ${s.status}

`; } } // 5. THE CRITICAL SYNTHESIS FUNCTION (Fixed Visibility) window.runSynthesisFlow = function() { if(!selectedSite) { alert("Please select a site on the map first."); return; } // Switch Page showPage('synthesis'); // Populate Synthesis UI document.getElementById('syn-site-title').innerText = selectedSite.name; document.getElementById('syn-links').innerHTML = ` ↗ NIAH APPRAISAL ↗ OPW DATA `; const cards = document.getElementById('syn-cards'); cards.innerHTML = `

AUTOMATED COLLISION

Precedent: ${selectedSite.name}

Logic: The ${selectedSite.style} morphology combined with a ${selectedSite.status} status requires immediate structural appraisal.

`; }; // 6. ${s.name}

PLAN: ${s.plan || 'Standard'}

`; document.getElementById('res-a').innerHTML = r(a); document.getElementById('res-b').innerHTML = r(b); }
